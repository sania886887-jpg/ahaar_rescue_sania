<script type="module">
  import { initializeApp } from "https://gstatic.com";
  import { getFirestore, collection, addDoc, onSnapshot, doc, updateDoc, query, orderBy, serverTimestamp } from "https://gstatic.com";

  const firebaseConfig = {
    apiKey: "AIzaSyB8wOrftXJWkybGGGjgQR1acT3a8VBtsYA",
    authDomain: "://firebaseapp.com",
    projectId: "aahaar-rescue",
    storageBucket: "aahaar-rescue.firebasestorage.app",
    messagingSenderId: "327571731574",
    appId: "1:327571731574:web:b03127a84c6a80eb871b57",
    measurementId: "G-NZ9WEHSV53"
  };

  const app = initializeApp(firebaseConfig);
  const db = getFirestore(app);

  // Tab Switcher
  window.showTab = (tabId, btn) => {
    document.querySelectorAll('.tab-section').forEach(el => el.classList.add('hidden'));
    document.querySelectorAll('.tab-btn').forEach(el => el.classList.remove('active'));
    document.getElementById(tabId).classList.remove('hidden');
    btn.classList.add('active');
  };

  // GPS Auto-Detect
  window.getGPS = () => {
    if (!navigator.geolocation) return alert("GPS not supported on this browser.");
    navigator.geolocation.getCurrentPosition(pos => {
      const mapsUrl = `https://google.com{pos.coords.latitude},${pos.coords.longitude}`;
      document.getElementById('location').value = mapsUrl;
      updateWhatsAppLink();
    }, () => alert("Please allow location permission in your browser."));
  };

  function updateWhatsAppLink() {
    const name = document.getElementById('donorName').value || "Donor";
    const food = document.getElementById('foodDetails').value || "Surplus Food";
    const loc = document.getElementById('location').value || "";
    const msg = encodeURIComponent(`Hello Aahaar-Rescue Team! Food pickup request from ${name}: ${food}. Location: ${loc}`);
    document.getElementById('waLink').href = `https://wa.me{msg}`;
  }
  document.getElementById('donorName').addEventListener('input', updateWhatsAppLink);
  document.getElementById('foodDetails').addEventListener('input', updateWhatsAppLink);
  updateWhatsAppLink();

  // Submit Request to Cloud Firestore
  window.submitRequest = async () => {
    const donorName = document.getElementById('donorName').value.trim();
    const phone = document.getElementById('phone').value.trim();
    const foodDetails = document.getElementById('foodDetails').value.trim();
    const location = document.getElementById('location').value.trim();

    if (!donorName || !foodDetails) return alert("Please enter Donor Name and Food Details.");

    try {
      await addDoc(collection(db, "food_requests"), {
        donorName, phone, foodDetails, location,
        status: "Pending",
        photoUrl: "",
        createdAt: serverTimestamp()
      });
      alert("✅ Request sent to Cloud! Switch to 'Live Donor Display' to track it.");
      document.getElementById('donorName').value = "";
      document.getElementById('phone').value = "";
      document.getElementById('foodDetails').value = "";
    } catch (e) {
      alert("Error: " + e.message + " (Make sure Firestore Rules are set to allow read, write)");
    }
  };

  // Confirm Pickup
  window.confirmPickup = async (id) => {
    await updateDoc(doc(db, "food_requests", id), { status: "Confirmed" });
  };

  // Compress Photo & Save Directly to Free Firestore Cloud
  window.uploadProof = (id, input) => {
    const file = input.files;
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (event) => {
      const img = new Image();
      img.onload = async () => {
        const canvas = document.createElement('canvas');
        const maxWidth = 600;
        const scale = maxWidth / img.width;
        canvas.width = maxWidth;
        canvas.height = img.height * scale;
        const ctx = canvas.getContext('2d');
        ctx.drawImage(img, 0, 0, canvas.width, canvas.height);
        const compressedBase64 = canvas.toDataURL('image/jpeg', 0.7);
        await updateDoc(doc(db, "food_requests", id), {
          status: "Delivered",
          photoUrl: compressedBase64
        });
        alert("📸 Pickup photo saved to Cloud and displayed live to Donor!");
      };
      img.src = event.target.result;
    };
    reader.readAsDataURL(file);
  };

  // Real-time Cloud Listener
  const q = query(collection(db, "food_requests"), orderBy("createdAt", "desc"));
  onSnapshot(q, (snapshot) => {
    const teamList = document.getElementById('teamList');
    const donorFeed = document.getElementById('donorFeed');
    teamList.innerHTML = "";
    donorFeed.innerHTML = "";

    if (snapshot.empty) {
      teamList.innerHTML = "<div class='card'>No requests yet. Submit one in Tab 1!</div>";
      donorFeed.innerHTML = "<div class='card'>No active donations yet.</div>";
      return;
    }

    snapshot.forEach(docSnap => {
      const data = docSnap.data();
      const id = docSnap.id;
      const mapLink = data.location && data.location.startsWith("http") ? `<a href="${data.location}" target="_blank">📍 Open Google Maps</a>` : (data.location || 'Not provided');

      teamList.innerHTML += `
        <div class="card">
          <span class="badge ${data.status}">${data.status}</span>
          <h3>${data.donorName} (${data.phone || 'No phone'})</h3>
          <p><strong>Food:</strong> ${data.foodDetails}</p>
          <p><strong>Location:</strong> ${mapLink}</p>
